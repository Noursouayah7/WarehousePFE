# Cell 4: helper loaders

def load_csv_file(filename):
    matches = glob.glob(os.path.join(DATA_DIR, '**', filename), recursive=True)
    if not matches:
        print(f'Missing CSV file: {filename}')
        return None
    path = matches[0]
    df = pd.read_csv(path)
    print(f'Loaded {filename}: shape={df.shape}')
    return df


def load_excel_file(filename, sheet_name=0, load_all_sheets=False):
    matches = glob.glob(os.path.join(DATA_DIR, '**', filename), recursive=True)
    if not matches:
        print(f'Missing Excel file: {filename}')
        return None
    path = matches[0]
    if load_all_sheets:
        sheets = pd.read_excel(path, sheet_name=None)
        print(f'Loaded {filename}: sheets={list(sheets.keys())}')
        return sheets
    df = pd.read_excel(path, sheet_name=sheet_name)
    print(f'Loaded {filename} (sheet={sheet_name}): shape={df.shape}')
    return df


def load_xml_file(filename):
    matches = glob.glob(os.path.join(DATA_DIR, '**', filename), recursive=True)
    if not matches:
        print(f'Missing XML file: {filename}')
        return None
    path = matches[0]
    try:
        df = pd.read_xml(path, xpath='.//record')
        print(f'Loaded {filename}: shape={df.shape}')
        return df
    except Exception as exc:
        print(f'Could not read {filename} with pandas.read_xml: {exc}')
        return None

# Cell 9: standardization helpers for the unified dataset
def snake_case(text):
    text = str(text).strip().lower()
    text = text.replace('%', ' pct ')
    text = text.replace('[', ' ').replace(']', ' ')
    text = text.replace('(', ' ').replace(')', ' ')
    text = text.replace('/', ' ')
    text = text.replace('-', ' ')
    text = text.replace(',', ' ')
    text = text.replace(':', ' ')
    text = text.replace('.', ' ')
    text = '_'.join(text.split())
    while '__' in text:
        text = text.replace('__', '_')
    return text.strip('_')


def standardize_frame(df):
    if df is None:
        return None
    out = df.copy()
    out.columns = [snake_case(col) for col in out.columns]
    out = out.loc[:, ~out.columns.duplicated()]
    out = out.dropna(axis=0, how='all')
    out = out.dropna(axis=1, how='all')
    unnamed_cols = [col for col in out.columns if col.startswith('unnamed')]
    if unnamed_cols:
        out = out.drop(columns=unnamed_cols)
    return out


def add_source_fields(df, source_dataset, record_type):
    if df is None:
        return None
    out = standardize_frame(df)
    out.insert(0, 'source_dataset', source_dataset)
    out.insert(1, 'record_type', record_type)
    return out


def coerce_numeric_columns(df, columns):
    if df is None:
        return None
    out = df.copy()
    for col in columns:
        if col in out.columns:
            out[col] = (
                out[col]
                .astype(str)
                .str.replace(',', '.', regex=False)
                .str.replace(' ', '', regex=False)
                .replace({'nan': np.nan, 'None': np.nan, '': np.nan})
            )
            out[col] = pd.to_numeric(out[col], errors='coerce')
    return out


def normalize_market_prices_sheet(sheets_dict):
    if not isinstance(sheets_dict, dict) or not sheets_dict:
        return None
    first_sheet_name = next(iter(sheets_dict))
    df = sheets_dict[first_sheet_name].copy()
    df = standardize_frame(df)
    df = add_source_fields(df, 'olive_oil_market_prices', 'market_prices')
    return df

# Cell 17: prepare features for price prediction
price_model_work_df = price_model_df.copy()

if 'date' in price_model_work_df.columns:
    price_model_work_df['date'] = pd.to_datetime(price_model_work_df['date'], errors='coerce')

price_model_work_df = price_model_work_df.dropna(subset=['price'])

target_column = 'price'
candidate_feature_columns = [
    'date_ordinal', 'year', 'month', 'cost', 'revenue', 'stock',
    'quantity_sold', 'product', 'country', 'source_dataset', 'record_type'
]
feature_columns = [col for col in candidate_feature_columns if col in price_model_work_df.columns]

model_df = price_model_work_df[feature_columns + [target_column]].copy()

for col in model_df.select_dtypes(include='object').columns:
    model_df[col] = model_df[col].astype('category').cat.codes

for col in model_df.columns:
    if col != target_column:
        if pd.api.types.is_numeric_dtype(model_df[col]):
            model_df[col] = model_df[col].fillna(model_df[col].median())

model_df = model_df.dropna(subset=[target_column]).reset_index(drop=True)

print('Model dataframe shape:', model_df.shape)
display(model_df.head(10))

# Cell 18: define train/test sets for the price model
X = model_df.drop(columns=[target_column])

if 'date_ordinal' in X.columns:
    X = X.sort_values(by='date_ordinal')
    y = y.loc[X.index]

    ('scaler', StandardScaler()),
    ('model', LinearRegression())
])

    n_estimators=300,
    random_state=42,
    max_depth=None,
    min_samples_split=2,
    min_samples_leaf=1,
    n_jobs=-1
)

random_forest_model.fit(X_train, y_train)

rf_predictions = random_forest_model.predict(X_test)

def regression_metrics(y_true, y_pred, model_name):
    mae = mean_absolute_error(y_true, y_pred)
    rmse = mean_squared_error(y_true, y_pred, squared=False)
    r2 = r2_score(y_true, y_pred)
    print(f'{model_name} -> MAE: {mae:.4f} | RMSE: {rmse:.4f} | R2: {r2:.4f}')

regression_metrics(y_test, linear_predictions, 'Linear Regression')
regression_metrics(y_test, rf_predictions, 'Random Forest Regressor')
